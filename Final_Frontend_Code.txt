    <!-- 4. Main Application JavaScript -->
    <script>
        document.addEventListener('DOMContentLoaded', () => {

            // --- Configuration ---
            const API_BASE_URL = 'http://127.0.0.1:8000/api/products';
            const OPEN_FOOD_FACTS_URL = 'https://world.openfoodfacts.org/api/v2/product';

            // --- Global State & Selectors ---
            const productForm = document.getElementById('add-product-form');
            const addProductBtn = document.getElementById('add-product-btn');
            const productListBody = document.getElementById('product-list-body');
            const searchInput = document.getElementById('search-input');
            const codeReader = new ZXing.BrowserMultiFormatReader();

            // Form inputs
            const nameInput = document.getElementById('form-name');
            const barcodeInput = document.getElementById('form-barcode');
            const expiryDateInput = document.getElementById('form-expiry-date');
            const formLoading = document.getElementById('form-loading');

            // Scanner modal elements
            const scannerModal = document.getElementById('scanner-modal');
            const openScannerBtn = document.getElementById('open-scanner-btn');
            const closeScannerBtn = document.getElementById('close-scanner-btn');

            // Stat card elements
            const statTotal = document.getElementById('stat-total');
            const statExpiringSoon = document.getElementById('stat-expiring-soon');
            const statExpired = document.getElementById('stat-expired');

            // Notification elements
            const notificationBanner = document.getElementById('notification-banner');
            const allowNotificationsBtn = document.getElementById('allow-notifications-btn');
            const notificationBannerText = document.getElementById('notification-banner-text');


            // --- Core Functions ---

            /**
             * Universal fetch wrapper for handling API requests and errors
             */
            async function apiRequest(url, options = {}) {
                try {
                    const response = await fetch(url, options);
                    if (!response.ok) {
                        // Try to get error message from server
                        let errorText = `HTTP error! Status: ${response.status}`;
                        try {
                            const errorData = await response.json();
                            errorText = `Error: ${JSON.stringify(errorData)}`;
                        } catch (e) { /* ignore json parse error */ }
                        throw new Error(errorText);
                    }
                    if (response.status === 204) return null; // No Content
                    return await response.json();
                } catch (err) {
                    console.error('API Request Error:', err);
                    throw err; // Re-throw the error to be caught by the caller
                }
            }

            /**
             * Calculates the difference in days between today and an expiry date.
             */
            function getDaysUntilExpiry(expiryDateString) {
                const expiryDate = new Date(expiryDateString + 'T00:00:00');
                const today = new Date();
                today.setHours(0, 0, 0, 0); // Normalize today's date

                const diffTime = expiryDate - today;
                return Math.ceil(diffTime / (1000 * 60 * 60 * 24));
            }

            /**
             * Renders the product list table
             */
            function renderProductList(products) {
                productListBody.innerHTML = ''; // Clear the list

                if (products.length === 0) {
                    productListBody.innerHTML = '<tr><td colSpan="5" class="p-4 text-center text-gray-500">No products found. Add one above!</td></tr>';
                    return;
                }

                products.forEach(product => {
                    const daysLeft = getDaysUntilExpiry(product.expiry_date);

                    let rowClass = 'bg-white hover:bg-gray-50';
                    let statusText = `${daysLeft} days left`;
                    let statusColor = 'text-green-600';

                    if (daysLeft < 0) {
                        rowClass = 'bg-red-100 hover:bg-red-200';
                        statusText = `Expired ${Math.abs(daysLeft)} days ago`;
                        statusColor = 'text-red-600';
                    } else if (daysLeft <= 7) {
                        rowClass = 'bg-yellow-100 hover:bg-yellow-200';
                        statusText = `${daysLeft} days left`;
                        statusColor = 'text-yellow-700';
                    }

                    const tr = document.createElement('tr');
                    tr.className = `transition-colors duration-200 ${rowClass}`;
                    tr.innerHTML = `
                        <td class="p-3 text-sm text-gray-700 font-medium">${product.name}</td>
                        <td class="p-3 text-sm text-gray-700 font-mono">${product.barcode}</td>
                        <td class="p-3 text-sm text-gray-700">${product.expiry_date}</td>
                        <td class="p-3 text-sm font-semibold ${statusColor}">${statusText}</td>
                        <td class="p-3">
                            <button data-id="${product.id}" class="delete-btn text-red-500 hover:text-red-700 font-semibold">
                                Delete
                            </button>
                        </td>
                    `;
                    productListBody.appendChild(tr);
                });
            }

            /**
             * Updates the dashboard stat cards
             */
            function renderDashboardStats(stats) {
                statTotal.textContent = stats.total_products;
                statExpiringSoon.textContent = stats.expiring_soon;
                statExpired.textContent = stats.expired;
            }

            /**
             * Fetches all data (list and stats) and renders the UI
             */
            async function loadAndRenderAllData() {
                const searchTerm = searchInput.value;
                try {
                    // Fetch list and stats in parallel
                    const [products, stats] = await Promise.all([
                        apiRequest(`${API_BASE_URL}/?search=${searchTerm}`),
                        apiRequest(`${API_BASE_URL}/stats/`)
                    ]);

                    renderProductList(products);
                    renderDashboardStats(stats);

                } catch (err) {
                    console.error("Error fetching data:", err);
                    productListBody.innerHTML = `
                        <tr>
                            <td colSpan="5" class="p-4 text-center text-red-700 bg-red-100">
                                <strong>Connection Error!</strong><br>
                                Could not connect to the Python server at ${API_BASE_URL}.<br>
                                Please ensure it is running and you are running this page from http://localhost:5500.
                            </td>
                        </tr>`;
                }
            }

            /**
             * Handles the product form submission.
             */
            async function handleAddProduct(e) {
                e.preventDefault();
                addProductBtn.disabled = true;
                addProductBtn.textContent = 'Adding...';

                const newProduct = {
                    name: nameInput.value,
                    barcode: barcodeInput.value,
                    expiry_date: expiryDateInput.value,
                    notified: false
                };

                try {
                    await apiRequest(`${API_BASE_URL}/`, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(newProduct)
                    });

                    productForm.reset();
                    await loadAndRenderAllData(); // Refresh UI

                } catch (err) {
                    alert(`Failed to add product: ${err.message}`);
                } finally {
                    addProductBtn.disabled = false;
                    addProductBtn.textContent = 'Add Product to Inventory';
                }
            }

            /**
             * Handles clicks on the product list (for deleting).
             */
            async function handleListClick(e) {
                if (e.target.classList.contains('delete-btn')) {
                    const id = e.target.getAttribute('data-id');
                    if (confirm('Are you sure you want to delete this product?')) {
                        try {
                            await apiRequest(`${API_BASE_URL}/${id}/`, {
                                method: 'DELETE'
                            });
                            await loadAndRenderAllData(); // Refresh UI
                        } catch (err) {
                            alert(`Failed to delete product: ${err.message}`);
                        }
                    }
                }
            }

            // --- Barcode Scanner Functions ---

            /**
             * Opens the barcode scanner modal.
             */
            function openScanner() {
                scannerModal.classList.remove('hidden');
                codeReader.decodeFromVideoDevice(undefined, 'scanner-video', (result, err) => {
                    if (result) {
                        barcodeInput.value = result.getText();
                        closeScanner();
                        handleBarcodeLookup(result.getText());
                    }
                    if (err && !(err instanceof ZXing.NotFoundException)) {
                        console.error(err);
                        alert(`Camera Error: ${err.message}`);
                        closeScanner();
                    }
                }).catch(err => {
                    console.error("Scanner init failed:", err);
                    alert("Could not initialize camera. Please grant permission.");
                    closeScanner();
                });
            }

            /**
             * Closes the barcode scanner modal.
             */
            function closeScanner() {
                scannerModal.classList.add('hidden');
                codeReader.reset(); // Stop the video stream
            }

            /**
             * Looks up a barcode on the Open Food Facts API
             */
            async function handleBarcodeLookup(barcode) {
                if (!barcode) return;

                nameInput.value = ''; // Clear name field
                formLoading.classList.remove('hidden'); // Show "Searching..."

                try {
                    const data = await apiRequest(`${OPEN_FOOD_FACTS_URL}/${barcode}.json`);

                    if (data.status === 1 && data.product?.product_name) {
                        const productName = data.product.product_name;
                        nameInput.value = productName; // Set product name
                    } else {
                        console.log(`Product with barcode ${barcode} was not found in Open Food Facts.`);
                        nameInput.placeholder = 'Product not found. Please enter manually.';
                    }

                } catch (err) {
                    console.error("Open Food Facts lookup error:", err);
                    nameInput.placeholder = 'Error finding product. Please enter manually.';
                } finally {
                    formLoading.classList.add('hidden'); // Hide "Searching..."
                }
            }


            // --- Notification Logic ---

            /**
             * Asks for permission to send notifications.
             */
            function askNotificationPermission() {
                if (!('Notification' in window)) {
                    notificationBannerText.textContent = "This browser does not support desktop notifications.";
                    notificationBanner.classList.remove('hidden');
                    return;
                }

                if (Notification.permission === 'granted') {
                    notificationBanner.classList.add('hidden');
                } else if (Notification.permission === 'denied') {
                    notificationBannerText.textContent = "Notifications are blocked. Please enable them in your browser settings.";
                    allowNotificationsBtn.classList.add('hidden');
                    notificationBanner.classList.remove('hidden');
                } else {
                    notificationBanner.classList.remove('hidden');
                }
            }

            /**
             * Handles the "Allow" button click for notifications.
             */
            function handleAllowNotifications() {
                Notification.requestPermission().then(permission => {
                    if (permission === 'granted') {
                        notificationBanner.classList.add('hidden');
                        new Notification('Notifications Enabled!', {
                            body: 'You will now receive expiry alerts.',
                            icon: 'https://placehold.co/192x192/6366f1/white?text=✓' // Simple checkmark icon
                        });
                    }
                });
            }

            /**
             * Checks for expiring products and sends notifications.
             */
            async function checkExpiryNotifications() {
                if (Notification.permission !== 'granted') {
                    return; // Don't check if we don't have permission
                }

                try {
                    const expiringProducts = await apiRequest(`${API_BASE_URL}/expiring_for_notification/`);

                    expiringProducts.forEach(product => {
                        const daysLeft = getDaysUntilExpiry(product.expiry_date);
                        new Notification('Product Expiring Soon!', {
                            body: `${product.name} (Barcode: ${product.barcode}) is expiring in ${daysLeft} ${daysLeft === 1 ? 'day' : 'days'}.`,
                            icon: 'https://placehold.co/192x192/f59e0b/white?text=!' // Simple warning icon
                        });
                    });
                } catch (err) {
                    console.error("Error checking for notifications:", err);
                }
            }

            // --- Event Listeners & Initialization ---

            // Form and list listeners
            productForm.addEventListener('submit', handleAddProduct);
            productListBody.addEventListener('click', handleListClick);
            searchInput.addEventListener('input', loadAndRenderAllData); // Re-fetch on search

            // Scanner listeners
            openScannerBtn.addEventListener('click', openScanner);
            closeScannerBtn.addEventListener('click', closeScanner);

            // Notification listeners
            allowNotificationsBtn.addEventListener('click', handleAllowNotifications);

            // Initial setup
            loadAndRenderAllData(); // Load data on page start
            
            // Notification setup
            askNotificationPermission();
            checkExpiryNotifications(); // Check once on load
            setInterval(checkExpiryNotifications, 60000); // Check every 60 seconds

        });
    </script>
